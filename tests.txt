






















10 COMPLETE JAVA EXCEPTION TEST PROGRAMS
Each example is intended to be pasted and run separately in your Java Exception Handler webpage.
Each uses the existing EH.runTask() recovery method.

============================================================
1. ArithmeticException — Division by Zero
============================================================
public class Main {
    public static void main(String[] args) {
        EH.runTask("Arithmetic Exception", () -> {
            int a = 10;
            int b = 0;
            int result = a / b;
            System.out.println(result);
        });

        System.out.println("Program continued");
    }
}

9. UnsupportedOperationException
============================================================

public class Main {
    public static void main(String[] args) {
        EH.runTask("Unsupported Operation Exception", () -> {
            List<Integer> numbers = Arrays.asList(10, 20, 30);

        });

        System.out.println("Program continued");
    }
}


============================================================
10. ConcurrentModificationException
============================================================
import java.util.ArrayList;


public class Main {
    public static void main(String[] args) {
        EH.runTask("Concurrent Modification Exception", () -> {
            ArrayList<Integer> numbers = new ArrayList<>();
            numbers.add(10);
            numbers.add(20);
            numbers.add(30);

            for (Integer number : numbers) {
                numbers.remove(number);
            }
        });

        System.out.println("Program continued");
    }
}
